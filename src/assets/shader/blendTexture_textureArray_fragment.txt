precision highp float;
precision highp int;
precision highp sampler2DArray;
precision highp sampler2D;

uniform sampler2DArray layers;
uniform sampler2D depth;

uniform int numImages;

uniform float showDepth;
uniform float minDepth;
uniform float maxDepth;
uniform float interpolateColor;
uniform float applyCalibration;
uniform float doOverrideDepth;
uniform float depthOverrideValue;
uniform float useRedChannelOnly;

uniform vec2 viewportSize;
uniform vec3 scale;
uniform vec3 translate;
uniform vec2 originalSize;

in vec2 vUv;
out vec4 outColor;

// conversion from Linear color space to sRGB
vec4 fromLinear(vec4 linearRGB)
{
	bvec4 cutoff = lessThan(linearRGB, vec4(0.0031308));
	vec4 higher = vec4(1.055)*pow(linearRGB, vec4(1.0/2.4)) - vec4(0.055);
	vec4 lower = linearRGB * vec4(12.92);

	return mix(higher, lower, cutoff);
}

// conversion from Linear color space to sRGB
vec4 toLinear(vec4 sRGB)
{
	bvec4 cutoff = lessThan(sRGB, vec4(0.04045));
	vec4 higher = pow((sRGB + vec4(0.055))/vec4(1.055), vec4(2.4));
	vec4 lower = sRGB/vec4(12.92);

	return mix(higher, lower, cutoff);
}

vec4 computeColorDynamic(vec2 texCoordsDepth, vec2 texCoords)
{	
	vec4 imageFront = texture(layers, vec3(texCoords, 1));
	vec4 imageBack = texture(layers, vec3(texCoords, 0));
	
	float depth = texture2D(depth, texCoordsDepth).x;
	// uncomment for testing:
	// depth = 0.4;

	if (doOverrideDepth > 0.5) {
		depth = 1.0 - depthOverrideValue;
	}

	depth = clamp(depth, minDepth, maxDepth);

	float factor = 1.0 / (maxDepth - minDepth);
	depth = 1.0 - ((depth - minDepth) * factor);

	if (showDepth > 0.5)
		return vec4(depth, depth, depth, 1.0);

	float layerWidth = 1.0 / float(numImages - 1);
	float layer = 0.0;

	int i = int(floor(depth / layerWidth));
	i = clamp(i, 0, numImages-1);

	imageFront = texture(layers, vec3(texCoords, i));
	imageBack = texture(layers, vec3(texCoords, i+1));
	layer = float(i); 

	float blend = (depth - (layer * layerWidth)) / layerWidth;

	vec4 result;

	if (interpolateColor > 0.5)
	{
		result = blend * imageBack + (1.0 - blend) * imageFront;
	}
	else
	{
		result = (depth > 0.95) ? imageBack : imageFront;
	}	

	vec4 converted = fromLinear(result);

	return converted;
}

float computeCalibratedPosition(float vpSize, float orgSize, float scl, float trans, float texcoord) {
	float factor = vpSize / orgSize;
	float calibrated = ((texcoord * factor)) / scl - trans / vpSize;
	return calibrated;
}

void main() {
	vec2 coords = vec2(mod(vUv.x, 1.0), mod(1.0 - vUv.y, 1.0));

	if (applyCalibration > 0.5) {
		
		float calibX = computeCalibratedPosition(viewportSize.x, originalSize.x, scale.x, translate.x, coords.x);
		float calibY = computeCalibratedPosition(viewportSize.y, originalSize.y, scale.y, translate.y, coords.y);
		
		vec2 calibratedCoords =  vec2(mod(calibX, 1.0), mod(1.0 - calibY, 1.0));

		outColor = computeColorDynamic(calibratedCoords, coords);
	} else {		
	    outColor = computeColorDynamic(vUv, coords);
	}

	if (useRedChannelOnly > 0.5) {
    	outColor = vec4( outColor.rrr * 1.5, 1.0 );
	}
}
